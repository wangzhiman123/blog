{{- /* 首页排行榜板块（热门 / 冷门各一个）。 用法： {{ partial "custom/popular.html" (dict "Mode" "hot") }} {{ partial "custom/popular.html" (dict "Mode" "cold") }} Mode hot —— 阅读量从多到少，只列有阅读量的（0 阅读的进不来，否则冷门榜会全是 0） cold —— 阅读量从少到多，0 阅读的也算（那才是真没人看）；同阅读量按发布日期升序，老文在前 数据从哪来： data/pv.json 由 tools/fetch-pv.py 在构建前抓取生成，结构： {"generated": "...", "pages": {"/posts/xxx/": 13, ...}} 里面记着全部文章的阅读量（含 0），所以冷门榜能取到 0 阅读的文章。 为什么板块会自动消失： 两边的配置 enable = false、data/pv.json 不存在（本地没跑脚本）、 或筛完一篇不剩时，整块都不输出。宁可不显示，也不显示空榜单。 路径怎么对上： pv.json 里存的是未编码的路径（中文目录名保持中文）， Hugo 的 .RelPermalink 是百分号编码过的，所以用 urls.Parse 的 .Path 解回来再比。 开关：config/_default/params.toml 的 [popular] / [unpopular] */ -}} {{- $mode := .Mode | default "hot" -}} {{- $isCold := eq $mode "cold" -}} {{- /* 配置项整段删掉时用空 dict 兜底，否则读 .enable 会报 nil 错误 */ -}} {{- $cfg := cond $isCold site.Params.unpopular site.Params.popular | default dict -}} {{- if ne $cfg.enable false -}} {{- $pv := dict -}} {{- with site.Data.pv -}} {{- with .pages -}} {{- $pv = . -}} {{- end -}} {{- end -}} {{- if $pv -}} {{- $mainSections := site.Params.mainSections | default (slice "posts") -}} {{- $items := slice -}} {{- range where site.RegularPages "Section" "in" $mainSections -}} {{- $key := (urls.Parse .RelPermalink).Path -}} {{- if isset $pv $key -}} {{- /* JSON 里的数字是 float64，直接 %d 会打印成 %!d(float64=19)，先转整数 */ -}} {{- $views := int (index $pv $key) -}} {{- /* 热门榜跳过 0 阅读；冷门榜相反，0 阅读正是要推荐的 */ -}} {{- if or $isCold (gt $views 0) -}} {{- /* 冷门榜的排序键：阅读量 + 发布日期拼成定长字符串。 直接 sort 只能按一个字段排，两次 sort 又会被后者打乱同分顺序， 所以拼成 "0000121697xxxxx" 这样的定长串，字典序即数值序。 */ -}} {{- $sortKey := printf "%010d%012d" $views (.Date.Unix) -}} {{- $items = $items | append (dict "Page" . "Views" $views "SortKey" $sortKey) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $items -}} {{- if $isCold -}} {{- $items = sort $items "SortKey" "asc" -}} {{- else -}} {{- $items = sort $items "Views" "desc" -}} {{- end -}} {{- $items = first ($cfg.count | default 5) $items -}}

{{- $cfg.title | default (cond $isCold "冷门文章" "热门文章") -}}

    {{- range $i, $item := $items }} {{- /* 冷门榜多半是 0 阅读的文章，连排五个「0 阅读」像数据出错，换成「暂无阅读」 */ -}} {{- $viewsText := cond (gt $item.Views 0) (printf "%d 阅读" $item.Views) "暂无阅读" -}}
  1. {{ add $i 1 }} {{ $item.Page.Title }} {{- /* data-short 给手机用：窄屏只显示数字，不显示「阅读」两字 */ -}} {{ $viewsText }}
    2. {{- end }}
{{- end -}} {{- end -}} {{- end -}}
